time_step_study

Time step study: how fine must the time step be to capture sudden road changes?

Runs a set of bumps and potholes with a fixed test car, compares each run with an accurate reference solution (see quarter_car/timestep.py) and saves these results in the results folder:

  • error_vs_time_step.png: how the error falls as the time step is reduced.
  • required_time_step.md: the time step each road and speed needs for 1%.
  • restart_road_readings.png and restart_acceleration_error.png: RK4 with and without restarts at the edges on the hardest pothole, same 1 ms step.
  • program_time_step_check.md: the program's 1 ms step on every road and speed.
  • input_range_check.md: the program's 1 ms step at the extremes of every input range.

Run from the project folder (takes a few minutes)::

python time_step_study.py

 72def observed_order(steps, errors, floor=1e-8):
 73    """Observed order p = log(E1 / E2) / log(h1 / h2) from the two finest steps.
 74
 75    Steps with errors below ``floor`` are skipped, because there the error is
 76    set by the accuracy of the reference and round-off, not by the step.
 77    """
 78    usable = np.asarray(errors) > floor
 79    h, e = np.asarray(steps)[usable], np.asarray(errors)[usable]
 80    if len(h) < 2:
 81        return float("nan")
 82    return float(np.log(e[-2] / e[-1]) / np.log(h[-2] / h[-1]))

103def convergence_figure():
104    """Error against time step at 40 km/h for a smooth road change and a sudden one."""
105    names = ["Speed hump, 75 mm x 3.7 m", "Medium pothole, 50 mm x 0.6 m"]
106    speed = 40 / 3.6
107    fig, axes = plt.subplots(1, 2, figsize=(8, 3.8), sharey=True)
108    for ax, name in zip(axes, names):
109        road = ROADS[name]
110        reference = reference_solution(CAR, road, speed)
111        orders = []
112        for method, (step, split) in METHODS.items():
113            steps = [h for h in FIGURE_STEPS if step is rk4_step or h < 7e-3]   # Euler is unstable above 7.2 ms
114            result = step_study(CAR, road, speed, steps, step, split, reference)
115            errors = np.maximum(result["history"], 1e-10)   # keep exact (zero) errors on the log axis
116            ax.loglog(1e3 * np.array(steps), 100 * errors, color=COLOURS[method], marker=MARKERS[method],
117                      ms=5, lw=1.8, label=method)
118            orders.append(f"{method}: {observed_order(steps, result['history']):.1f}")
119        ax.axhline(100 * TOL, color=MUTED, lw=1)
120        ax.text(0.045, 100 * TOL * 1.4, "1% error", color=SECONDARY, fontsize=8,
121                bbox=dict(facecolor="white", edgecolor="none", pad=1))
122        ax.set_title(name, fontsize=9)
123        ax.set_xlabel("Time step h [ms]")
124        ax.text(0.97, 0.04, "Observed order\n" + "\n".join(orders), transform=ax.transAxes,
125                fontsize=7, color=SECONDARY, ha="right", va="bottom")
126        style(ax)
127        print(f"  {name}: observed order " + ", ".join(orders))
128    axes[0].set_ylabel("Largest error in body acceleration\nand tyre force [% of peak]")
129    axes[0].set_ylim(1e-7, 1e3)
130    handles, labels = axes[0].get_legend_handles_labels()
131    fig.legend(handles, labels, loc="upper center", bbox_to_anchor=(0.5, 0.93), ncol=3,
132               fontsize=8, frameon=False)
133    fig.suptitle("Error against time step at 40 km/h", color=INK, fontsize=11)
134    fig.tight_layout(rect=(0, 0, 1, 0.9))
135    fig.savefig(RESULTS / "error_vs_time_step.png", dpi=200)
136    plt.close(fig)

139def required_step_table():
140    """Time step for the solver to follow the motion within 1%, for every road and speed."""
141    lines = ["| Road | Method | " + " | ".join(f"{v} km/h" for v in SPEEDS_KMH) + " |",
142             "|---|---|" + "---|" * len(SPEEDS_KMH)]
143    for name, road in ROADS.items():
144        cells = {"RK4": [], "RK4, restarted at edges": []}
145        for kmh in SPEEDS_KMH:
146            reference = reference_solution(CAR, road, kmh / 3.6)
147            for method in cells:
148                step, split = METHODS[method]
149                result = step_study(CAR, road, kmh / 3.6, STEPS, step, split, reference)
150                h, _ = required_step(result, TOL, measure="history")
151                cells[method].append("< 0.1" if h is None else f"{1e3 * h:g}")
152        for method, values in cells.items():
153            lines.append(f"| {name} | {method} | " + " | ".join(values) + " |")
154        print(f"  {name}: done")
155    table = "\n".join(lines)
156    (RESULTS / "required_time_step.md").write_text(
157        "Largest time step [ms] for which the body acceleration and tyre force stay within 1% "
158        "of the reference at every step\n\n"
159        + table + "\n")
160    print(table)

163def restart_figures():
164    """Show RK4 with and without restarts at the edges on the hardest pothole.
165
166    The hardest pothole is the deepest and shortest one at the top speed. Both
167    runs use the program's 1 ms step, so the only difference is the restart.
168    Without restarting, the step that ends on the near edge and the step that
169    crosses the far edge each read the road on both sides of the edge, which
170    puts an error of about 2% into the body acceleration. Restarting keeps
171    every step on one side of an edge. Saves two figures: where each method
172    reads the road, and the error in body acceleration that results.
173    """
174    road, speed, h = Pothole(0.15, 0.3), 110 / 3.6, TIME_STEP
175    reference = reference_solution(CAR, road, speed)
176    road_height = road_input(road, speed)
177    near_edge, far_edge = 0.3, 0.3 + road.length / speed         # times the tyre meets each edge [s]
178    title = "110 km/h over a 150 mm deep, 0.3 m long pothole"
179    ring = "#fcfcfb"
180    methods = {   # restarted at edges?: (name, colour, marker style)
181        False: ("not restarted", COLOURS["RK4"], dict(marker="s", ms=7.5, mfc=ring, mec=COLOURS["RK4"], mew=1.4)),
182        True: ("restarted at edges", COLOURS["RK4, restarted at edges"], dict(marker="^", ms=5.5, mec=ring, mew=0.8)),
183    }
184    runs = {restarted: simulate_road(CAR, road, speed, h=h, split_at_edges=restarted)[:2] for restarted in methods}
185
186    # Figure 1: the moments each method reads the road around the pothole.
187    zoom = (0.297, 0.313)                                         # time window shown [s]
188    t_zoom = np.arange(zoom[0], zoom[1], 1e-6)
189    fig, ax = plt.subplots(figsize=(7, 3.6))
190    ax.plot(1e3 * t_zoom, 1e3 * road_height(t_zoom), color=MUTED, lw=1.5, label="Road under the tyre")
191    for restarted, (name, colour, shape) in methods.items():
192        t = runs[restarted][0]
193        nudge = 1e-12 if restarted else 0.0     # a restarted run reads just inside each segment
194        reads = np.concatenate([t[:-1] + nudge, (t[:-1] + t[1:]) / 2, t[1:] - nudge])   # start, middle, end
195        reads = reads[(reads >= zoom[0]) & (reads <= zoom[1])]
196        ax.plot(1e3 * reads, 1e3 * road_height(reads), ls="none", color=colour, **shape,
197                label=f"RK4 readings, 1 ms step, {name}")
198    label = "Step that reads both sides of an edge (not restarted)"
199    t = runs[False][0]
200    for start, end in zip(t[:-1], t[1:]):
201        if zoom[0] <= start < zoom[1] and np.ptp(road_height(np.array([start, (start + end) / 2, end]))) > 0:
202            ax.axvspan(1e3 * start, 1e3 * end, color=COLOURS["RK4"], alpha=0.18, lw=0, label=label)
203            label = "_nolegend_"                 # one legend entry for all shaded steps
204    ax.set_title(f"Where RK4 reads the road, {title}", fontsize=10)
205    ax.set_xlabel("Time [ms]")
206    ax.set_ylabel("Road height [mm]")
207    ax.set_xlim(1e3 * zoom[0], 1e3 * zoom[1])
208    ax.set_ylim(-175, 30)
209    ax.legend(fontsize=7.5, frameon=False, loc="center")
210    style(ax)
211    fig.tight_layout()
212    fig.savefig(RESULTS / "restart_road_readings.png", dpi=200)
213    plt.close(fig)
214
215    # Figure 2: the error in body acceleration that results.
216    view = (0.28, 0.46)                                           # time window shown [s]
217    t_run = np.arange(0.0, far_edge + 2.0, 1e-5)
218    accel_peak = np.abs(body_acceleration(CAR, reference.states(t_run))).max()
219    fig, ax = plt.subplots(figsize=(7, 3.6))
220    ax.axvspan(1e3 * near_edge, 1e3 * far_edge, color=MUTED, alpha=0.15, lw=0)
221    ax.text(1e3 * far_edge + 1.5, 0.95, "Pothole", transform=ax.get_xaxis_transform(),
222            fontsize=7.5, color=SECONDARY, va="top")
223    for restarted, (name, colour, _) in methods.items():
224        t, Z = runs[restarted]
225        error, _ = run_errors(CAR, t, Z, reference)
226        error_text = "< 0.01%" if error < 1e-4 else f"{100 * error:.1f}%"
227        accel_error = body_acceleration(CAR, Z) - body_acceleration(CAR, reference.states(t))
228        ax.plot(1e3 * t, 100 * accel_error / accel_peak, color=colour, lw=1.8,
229                label=f"RK4, 1 ms step, {name} (error {error_text})")
230        print(f"  RK4, 1 ms step, {name}: error {error_text}")
231    ax.set_title(f"Error in body acceleration, {title}", fontsize=10)
232    ax.set_xlabel("Time [ms]")
233    ax.set_ylabel("Error [% of peak]")
234    ax.set_xlim(1e3 * view[0], 1e3 * view[1])
235    ax.legend(fontsize=7.5, frameon=False, loc="lower right")
236    style(ax)
237    fig.tight_layout()
238    fig.savefig(RESULTS / "restart_acceleration_error.png", dpi=200)
239    plt.close(fig)

242def check_program_step():
243    """Check the program's time step (RK4 restarted at edges) on every road and speed.
244
245    Unlike the tables above, this also checks the peak values the program
246    reports, which can be missed between steps if the step is too coarse.
247    """
248    worst_history, worst_peaks = 0.0, 0.0
249    for name, road in ROADS.items():
250        for kmh in SPEEDS_KMH + (110,):
251            reference = reference_solution(CAR, road, kmh / 3.6)
252            t, Z, _ = simulate_road(CAR, road, kmh / 3.6, h=TIME_STEP, split_at_edges=True)
253            history, peaks = run_errors(CAR, t, Z, reference)
254            worst_history, worst_peaks = max(worst_history, history), max(worst_peaks, peaks)
255    text = (f"Program time step h = {1e3 * TIME_STEP:g} ms (RK4 restarted at edges), every road at "
256            f"{', '.join(str(v) for v in SPEEDS_KMH + (110,))} km/h: worst motion error "
257            f"{100 * worst_history:.2f}%, worst peak error {100 * worst_peaks:.2f}%")
258    (RESULTS / "program_time_step_check.md").write_text(text + "\n")
259    print(text)

262def check_input_ranges():
263    """Check the program's time step at the extremes of every input range.
264
265    Every combination of the lowest and highest body mass, spring stiffness,
266    damping ratio and tyre stiffness (16 cars) is run over six extreme roads:
267    the deepest bump and pothole at their shortest and the top speed, at their
268    longest and the lowest speed, and the shallowest of each at the top speed.
269    A run passes if both the motion and the peaks are within 1%.
270    """
271    size, length, speed = (limits[3:] for limits in (*ROAD_INPUTS, SPEED_INPUT))   # (lowest, highest)
272    roads = [("pothole", size[1], length[0], speed[1]), ("bump", size[1], length[0], speed[1]),
273             ("pothole", size[1], length[1], speed[0]), ("bump", size[1], length[1], speed[0]),
274             ("pothole", size[0], length[0], speed[1]), ("bump", size[0], length[1], speed[1])]
275    rows = []
276    for car_limits in itertools.product(*[(low, high) for *_, low, high in CAR_INPUTS]):
277        values = {key: value for (key, *_), value in zip(CAR_INPUTS, car_limits)}
278        car = build_car(values)
279        for kind, height, road_length, kmh in roads:
280            road = build_road(kind, {"height_or_depth": height, "length": road_length})
281            reference = reference_solution(car, road, kmh / 3.6)
282            t, Z, _ = simulate_road(car, road, kmh / 3.6, h=TIME_STEP, split_at_edges=True)
283            history, peaks = run_errors(car, t, Z, reference)
284            rows.append((max(history, peaks), history, peaks, car_limits,
285                         f"{height:g} mm x {road_length:g} m {kind}", kmh))
286    passed = sum(worst <= TOL for worst, *_ in rows)
287    summary = (f"Program time step h = {1e3 * TIME_STEP:g} ms (RK4 restarted at edges) at the extremes of "
288               f"every input range: {len(rows) // len(roads)} cars (every combination of the lowest and "
289               f"highest body mass, spring stiffness, damping ratio and tyre stiffness) over {len(roads)} "
290               f"extreme roads. Runs with both the motion and the peaks within 1% of the reference: "
291               f"{passed} of {len(rows)}. Worst error: {100 * max(rows)[0]:.2f}%.")
292    lines = ["| Body mass [kg] | Spring [kN/m] | Damping ratio | Tyre [kN/m] | Road | Speed [km/h] "
293             "| Motion error | Peak error |", "|---|---|---|---|---|---|---|---|"]
294    for _, history, peaks, (mass, spring, zeta, tyre), road_name, kmh in sorted(rows, reverse=True)[:5]:
295        lines.append(f"| {mass:g} | {spring:g} | {zeta:g} | {tyre:g} | {road_name} | {kmh:g} | "
296                     f"{100 * history:.2f}% | {100 * peaks:.2f}% |")
297    (RESULTS / "input_range_check.md").write_text(summary + "\n\nThe five worst runs:\n\n"
298                                                  + "\n".join(lines) + "\n")
299    print(summary)

302def main():
303    """Run every part of the study and save the results."""
304    RESULTS.mkdir(exist_ok=True)
305    parts = [("Context", print_context), ("Reference check", check_reference),
306             ("Convergence figure", convergence_figure), ("Required time step", required_step_table),
307             ("Restart figures", restart_figures),
308             ("Check of the program's time step", check_program_step),
309             ("Check at the extremes of the input ranges", check_input_ranges)]
310    for title, part in parts:
311        start = time.perf_counter()
312        print(f"\n== {title}")
313        part()
314        print(f"   ({time.perf_counter() - start:.0f} s)")
315    print(f"\nSaved figures and tables in {RESULTS}")
