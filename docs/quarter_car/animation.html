quarter_car.animation

Animation of the quarter car driving over a road.

The left panel draws the same two-mass model that is simulated, laid out like the simplified system diagram: the body m2 sits on the suspension spring Ks and damper b, which sit on the wheel and axle m1, which sits on the tyre spring Kw touching the road. The car stays in the centre of the view and the road moves underneath it. If the wheel leaves the road, the tyre spring is drawn at its unloaded length with a gap below it.

The right panels show the road, wheel and body displacement and the tyre force against time, and the text underneath lists the inputs and key results.

The animation plays in real time by default. Each frame shows the moment of the simulation that matches the time on the clock since playback started, so if the computer draws frames slowly, frames are skipped rather than the animation slowing down. For smooth playback it also uses blitting: the axes, labels and time histories are drawn once, and only the moving parts are redrawn each frame.

 31# Drawing settings (only affect the picture, not the physics). Heights are at rest [m].

 44def spring_points(x_pos, bottom, top, coils=6, width=0.06):
 45    """Return x and y coordinates of a zig-zag spring between two heights."""
 46    ys = np.linspace(bottom, top, 2 * coils + 3)
 47    xs = np.full_like(ys, x_pos)
 48    xs[1:-1] += width * np.where(np.arange(1, len(ys) - 1) % 2, 1, -1)
 49    xs[1], xs[-2] = x_pos, x_pos            # straight ends
 50    return xs, ys

 53def damper_points(x_pos, bottom, top):
 54    """Return x and y coordinates of a damper (dashpot) between two heights.
 55
 56    The cylinder is fixed to the lower mass and the piston rod to the upper
 57    mass, so the piston moves into the cylinder as the suspension compresses.
 58    NaN values split the line into separate pieces.
 59    """
 60    base = bottom + 0.05                    # bottom of the cylinder
 61    piston = top - ROD_LEN                  # piston height
 62    w, nan = CYL_HALF_W, np.nan
 63    xs = [x_pos, x_pos, nan,                                  # stem
 64          x_pos - w, x_pos - w, x_pos + w, x_pos + w, nan,    # cylinder
 65          x_pos, x_pos, nan,                                  # rod
 66          x_pos - 0.8 * w, x_pos + 0.8 * w]                   # piston
 67    ys = [bottom, base, nan,
 68          base + CYL_LEN, base, base, base + CYL_LEN, nan,
 69          top, piston, nan,
 70          piston, piston]
 71    return xs, ys

 74class _ClosableAnimation(FuncAnimation):
 75    """FuncAnimation that stops cleanly when its window is closed.
 76
 77    Clicking the window's close button detaches the figure from the window
 78    straight away, but one more timer tick can still arrive before the window
 79    is destroyed (more likely on a slower computer). With blitting, that tick
 80    would redraw a figure that has no window and raise an AttributeError, so
 81    each tick first checks that the window is still there.
 82    """
 83
 84    def _step(self, *args):
 85        if self._fig.canvas.manager is None:      # the window has been closed
 86            self.event_source.stop()
 87            return False                          # tells the timer to drop this animation
 88        return super()._step(*args)

 91def describe_modes(car):
 92    """Return a one-line description of the natural frequencies and damping."""
 93    try:
 94        freqs, zetas = natural_modes(car)
 95    except ValueError:
 96        return "Natural frequencies: not shown, because a mode is overdamped"
 97    return (f"Natural frequencies: body {freqs[0]:.2f} Hz (damping {zetas[0]:.2f}), "
 98            f"wheel hop {freqs[1]:.1f} Hz (damping {zetas[1]:.2f})")

101def style(ax):
102    """Give an axes hairline gridlines, quiet axis lines and dark text."""
103    ax.grid(True, color=GRID, lw=0.6)
104    ax.set_axisbelow(True)
105    for spine in ax.spines.values():
106        spine.set_color("#c3c2b7")
107    ax.tick_params(colors=SECONDARY)

110def animate_road(car, road, speed, h=1e-3, lead_in_time=0.3, t_after=2.0,
111                 playback=1.0, fps=50, hold_frames=40):
112    """Simulate the car driving over a road, then play it as an animation.
113
114    The run uses RK4 restarted at every sudden change in the road (see
115    ``scenarios.simulate_road``). Opens a window and returns once the user
116    closes it. The animation repeats until then.
117
118    Parameters
119    ----------
120    car : QuarterCar
121        Model parameters.
122    road : Bump or Pothole
123        Road from ``quarter_car.road``.
124    speed : float
125        Vehicle speed [m/s].
126    h : float, optional
127        RK4 step size [s]. Default 1 ms.
128    lead_in_time : float, optional
129        Time before the tyre reaches the road feature [s]. Default 0.3.
130    t_after : float, optional
131        Time simulated after the tyre leaves the feature [s]. Default 2.0.
132    playback : float, optional
133        Animation speed as a fraction of real time. Default 1 (real time);
134        for example 0.25 plays at quarter speed.
135    fps : int, optional
136        Animation frames per second. Default 50.
137    hold_frames : int, optional
138        Frames to pause at the end before repeating. Default 40.
139
140    Returns
141    -------
142    dict
143        Key results from :func:`quarter_car.scenarios.road_summary`.
144    """
145    t, Z, r = simulate_road(car, road, speed, h=h, lead_in_time=lead_in_time,
146                            t_after=t_after, split_at_edges=True)
147    results = road_summary(car, t, Z, r)
148    force = tyre_force(car, Z, r)
149    static_load = (car.m1 + car.m2) * G
150    static_deflection = static_load / car.Kw   # how much the tyre is squashed at rest [m]
151
152    # Interpolate onto evenly spaced frame times so playback speed is exact.
153    frame_t = np.append(np.arange(0.0, t[-1], playback / fps), t[-1])
154    frame_x = np.interp(frame_t, t, Z[:, 0])
155    frame_y = np.interp(frame_t, t, Z[:, 1])
156    frame_r = road_input(road, speed, lead_in_time)(frame_t)
157
158    # ---- Figure layout: still parts are drawn once, moving parts are animated.
159    fig = plt.figure(figsize=(13, 6.5))
160    ax_car = fig.add_axes([0.06, 0.22, 0.42, 0.70])
161    ax_disp = fig.add_axes([0.56, 0.60, 0.41, 0.32])
162    ax_force = fig.add_axes([0.56, 0.22, 0.41, 0.30], sharex=ax_disp)
163
164    ax_car.set_xlim(-VIEW_HALF_WIDTH, VIEW_HALF_WIDTH)
165    ax_car.set_ylim(-0.25, M2_BASE + M2_H + 0.2)
166    ax_car.set_aspect("equal", adjustable="box")
167    ax_car.set_xlabel("Distance from the car [m]")
168    ax_car.set_ylabel("Height [m]")
169    ax_car.set_title(f"{3.6 * speed:.0f} km/h, {road.label}")
170
171    (road_line,) = ax_car.plot([], [], color=ROAD, lw=2, animated=True)
172    m1_box = ax_car.add_patch(Rectangle((0, M1_BASE), M1_W, M1_H, color=WHEEL, animated=True))
173    m2_box = ax_car.add_patch(Rectangle((0, M2_BASE), M2_W, M2_H, color=BODY, animated=True))
174    (tyre_spring,) = ax_car.plot([], [], color=INK, lw=1.5, animated=True)
175    (susp_spring,) = ax_car.plot([], [], color=INK, lw=1.5, animated=True)
176    (damper,) = ax_car.plot([], [], color=SECONDARY, lw=1.5, animated=True)
177    m1_label = ax_car.text(0, 0, r"$m_1$", ha="center", va="center", color="white", animated=True)
178    m2_label = ax_car.text(0, 0, r"$m_2$", ha="center", va="center", color="white", animated=True)
179    kw_label = ax_car.text(0, 0, r"$K_w$", va="center", animated=True)
180    ks_label = ax_car.text(0, 0, r"$K_s$", ha="right", va="center", animated=True)
181    b_label = ax_car.text(0, 0, r"$b$", va="center", animated=True)
182    time_text = ax_car.text(0.02, 0.97, "", transform=ax_car.transAxes, va="top", animated=True)
183
184    ax_disp.plot(t, 1e3 * r, color=ROAD, lw=1.5, label="Road r")
185    ax_disp.plot(t, 1e3 * Z[:, 0], color=WHEEL, lw=1.5, label="Wheel x (m1)")
186    ax_disp.plot(t, 1e3 * Z[:, 1], color=BODY, lw=1.5, label="Body y (m2)")
187    ax_disp.set_ylabel("Displacement [mm]")
188    ax_disp.legend(loc="upper right", fontsize=8, frameon=False)
189    ax_disp.set_title(f"Peak body displacement {1e3 * results['peak_body_displacement']:.1f} mm, "
190                      f"peak body acceleration {results['peak_body_acceleration']:.1f} m/s$^2$", fontsize=10)
191    plt.setp(ax_disp.get_xticklabels(), visible=False)
192
193    ax_force.plot(t, force / 1e3, color=TYRE, lw=1.5)
194    ax_force.axhline(static_load / 1e3, color=ROAD, lw=1)
195    ax_force.text(t[-1], static_load / 1e3, "static load ", ha="right", va="bottom", fontsize=8, color=SECONDARY)
196    ax_force.set_xlim(0, t[-1])
197    ax_force.set_xlabel("Time [s]")
198    ax_force.set_ylabel("Tyre force [kN]")
199    if results["lift_off_time"] is None:
200        ax_force.set_title(f"Tyre force stays above {results['min_tyre_force']:.0f} N, "
201                           "so the wheel keeps its grip", fontsize=10)
202    else:
203        ax_force.set_title(f"Warning: the wheel leaves the road for {1e3 * results['airborne_time']:.0f} ms "
204                           f"(first at t = {results['lift_off_time']:.2f} s)", fontsize=10, color=WARNING)
205    cursors = [ax.axvline(0, color=INK, lw=1, animated=True) for ax in (ax_disp, ax_force)]
206    for ax in (ax_disp, ax_force):
207        style(ax)
208
209    info = "\n".join([
210        f"Inputs: body mass {car.m2:.0f} kg, wheel mass {car.m1:.0f} kg (fixed), "
211        f"spring {car.Ks / 1e3:.1f} kN/m, damper {car.b:.0f} N s/m, tyre {car.Kw / 1e3:.0f} kN/m",
212        describe_modes(car),
213        f"Results: peak body acceleration {results['peak_body_acceleration']:.2f} m/s^2, "
214        f"peak suspension travel {1e3 * results['peak_suspension_travel']:.1f} mm, "
215        f"tyre force {results['min_tyre_force']:.0f} to {results['max_tyre_force']:.0f} N",
216    ])
217    fig.text(0.04, 0.04, info, fontsize=10, va="bottom")
218
219    clock = {"start": None}
220
221    def update(_frame):
222        """Move both masses, both springs, the damper and the road to the current moment."""
223        if clock["start"] is None:                # playback starts at the first frame
224            clock["start"] = time.perf_counter()
225        elapsed = time.perf_counter() - clock["start"]   # real time since playback started [s]
226        n = len(frame_t)
227        k = min(int(elapsed * fps) % (n + hold_frames), n - 1)   # frame for this moment, then hold
228
229        x, y, road_height = frame_x[k], frame_y[k], frame_r[k]
230        distance = speed * (frame_t[k] - lead_in_time)       # distance past the start of the feature
231        m1_bottom, m2_bottom = M1_BASE + x, M2_BASE + y
232        m1_top = m1_bottom + M1_H
233        # In the air the tyre spring hangs at its unloaded length, so its bottom is above the road.
234        tyre_bottom = max(road_height, x - static_deflection)
235
236        # The car stays at x = 0 and the road slides underneath it.
237        road_line.set_data(ROAD_X, road.profile(distance + ROAD_X))
238        m1_box.set_xy((-M1_W / 2, m1_bottom))
239        m2_box.set_xy((-M2_W / 2, m2_bottom))
240        tyre_spring.set_data(*spring_points(0.0, tyre_bottom, m1_bottom))   # Kw: road to m1
241        susp_spring.set_data(*spring_points(-SIDE, m1_top, m2_bottom))      # Ks: m1 to m2
242        damper.set_data(*damper_points(SIDE, m1_top, m2_bottom))            # b: m1 to m2
243
244        m1_label.set_position((0.0, m1_bottom + M1_H / 2))
245        m2_label.set_position((0.0, m2_bottom + M2_H / 2))
246        kw_label.set_position((0.10, (tyre_bottom + m1_bottom) / 2))
247        ks_label.set_position((-SIDE - 0.10, (m1_top + m2_bottom) / 2))
248        b_label.set_position((SIDE + 0.09, (m1_top + m2_bottom) / 2))
249
250        for cursor in cursors:
251            cursor.set_xdata([frame_t[k], frame_t[k]])
252        time_text.set_text(f"t = {frame_t[k]:.2f} s")   # kept short: long text is slow to redraw
253        return (road_line, m1_box, m2_box, tyre_spring, susp_spring, damper,
254                m1_label, m2_label, kw_label, ks_label, b_label, time_text, *cursors)
255
256    # Keep a reference to the animation while the window is open.
257    animation = _ClosableAnimation(fig, update, interval=1000 / fps, blit=True, cache_frame_data=False)
258    plt.show()
259    plt.close(fig)
260    return results

def damper_points(x_pos, bottom, top): 
54def damper_points(x_pos, bottom, top):
55    """Return x and y coordinates of a damper (dashpot) between two heights.
56
57    The cylinder is fixed to the lower mass and the piston rod to the upper
58    mass, so the piston moves into the cylinder as the suspension compresses.
59    NaN values split the line into separate pieces.
60    """
61    base = bottom + 0.05                    # bottom of the cylinder
62    piston = top - ROD_LEN                  # piston height
63    w, nan = CYL_HALF_W, np.nan
64    xs = [x_pos, x_pos, nan,                                  # stem
65          x_pos - w, x_pos - w, x_pos + w, x_pos + w, nan,    # cylinder
66          x_pos, x_pos, nan,                                  # rod
67          x_pos - 0.8 * w, x_pos + 0.8 * w]                   # piston
68    ys = [bottom, base, nan,
69          base + CYL_LEN, base, base, base + CYL_LEN, nan,
70          top, piston, nan,
71          piston, piston]
72    return xs, ys
